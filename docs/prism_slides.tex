\begin{frame}[label={sec:orgfd7a113}]{What we are doing}
\begin{itemize}
\item Quant psych: repeated measurements per person over time —
longitudinal data
\item Missing data is the norm: dropout, skipped waves
\item Most analyses need a completed dataset
\item prism: complete the data so its structure matches the model
\end{itemize}
\end{frame}

\begin{frame}[label={sec:orge6bf122},fragile]{FIML estimates — but returns no dataset}
 \begin{itemize}
\item FIML fits the model under MAR — but returns no completed dataset
\item \alert{Unconstrained imputation is not desirable:} \texttt{lavPredict} shrinks
variance 40–60\%, \texttt{MICE} and \texttt{missForest} drift the covariance — it
needs a constraint
\end{itemize}
\end{frame}

\begin{frame}[label={sec:org3570a17}]{The idea — project onto the FIML structure}
\begin{itemize}
\item FIML gives the target \((\hat{\mu}, \hat{\Sigma})\)
\item prism moves the imputed scatter onto that target — observed cells
untouched
\end{itemize}

\begin{frame}[label={sec:org5bb0403},fragile]{What prism takes in}
 \begin{itemize}
\item \alert{In:} incomplete data + \texttt{lavaan} model + initial imputation \(X^{(0)}\)
(e.g. \texttt{missForest})
\item \alert{Out:} completed data matching the FIML target — only missing cells move
\item \alert{Fallback:} no initial imputation? column means are used — worse
values, same structure
\end{itemize}

\begin{frame}[label={sec:orgca32b56}]{The objective — one equation, two forces}
\begin{itemize}
\item Move only the missing cells to minimize

$$ f(X) = \frac{1}{2N_{\mathrm{mis}}} \lVert M \odot (X - X^{(0)}) \rVert_{F}^{2} + \frac{\lambda_{\Sigma}}{2p^{2}} \lVert \operatorname{Cov}(X) - \Sigma_{\mathrm{FIML}} \rVert_{F}^{2} $$

\item \alert{First term} — stay close to the initial imputation
\item \alert{Second term} — pull the covariance to the FIML target
\item \(\lambda_{\Sigma}\) balances the two
\end{itemize}
\end{frame}

\begin{frame}[label={sec:orgb195e49},fragile]{Usage}
 \begin{verbatim}
library(prism)

model <- "
  i =~ 1*T1 + 1*T2 + 1*T3 + 1*T4
  s =~ 0*T1 + 1*T2 + 2*T3 + 3*T4"

imp <- prism_fiml(data = df, model = model,
                  lambda_sigma = 0.5)
attr(imp, "prism_diagnostics")   # convergence evidence
\end{verbatim}
\end{frame}

\begin{frame}[label={sec:orgee64611},fragile]{Results}
 \begin{itemize}
\item GCM + SEM; \(N\) 100–5000; 5–30\% missing; MAR/MNAR — distance to the
true covariance
\item GCM, Normal, MAR: FIML 1.35 · \texttt{lavPredict} 4.68 · prism 6.66 ·
\texttt{MICE} 6.70 · \texttt{missForest} 6.74
\item GCM, Normal, MNAR: FIML 3.48 · prism 4.00 · \texttt{missForest} 4.08 ·
\texttt{MICE} 4.40 · \texttt{lavPredict} 7.18
\item SEM, Normal: FIML 0.30 · \texttt{MICE} 0.44 · prism 0.61 · \texttt{missForest} 0.62 ·
\texttt{lavPredict} 1.45 — prism beats its starting imputation in every
setting
\end{itemize}
\end{frame}

\begin{frame}[label={sec:org49f0d67},fragile]{Multiple imputation — and the honest caveat}
 \begin{itemize}
\item \texttt{prism\_mi}: proper multiple imputation — fresh initializations +
parameter draws
\item prism completes data for everything FIML cannot feed; refitting on
prism data beats the initial imputation — still short of FIML
\end{itemize}
\end{frame}

\begin{frame}[label={sec:org563be16},fragile]{Take-aways}
 \begin{itemize}
\item prism projects an initial imputation onto the FIML-implied
\((\mu, \Sigma)\) — best completed-data covariance recovery in the grid
\item R package, MIT: \texttt{install\_github("xguot/prism")}
\end{itemize}
\end{frame}
